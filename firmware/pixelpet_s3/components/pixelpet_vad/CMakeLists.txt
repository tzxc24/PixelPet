set(ESP_SR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../tmp/esp-sr-reference")
if(CONFIG_PIXELPET_USE_ESP_SR_VAD)
    if(NOT IDF_TARGET STREQUAL "esp32s3")
        message(FATAL_ERROR "PixelPet's standalone ESP-SR VAD adapter targets ESP32-S3.")
    endif()
    set(VAD_ARCHIVE "${ESP_SR_ROOT}/lib/esp32s3/libesp_audio_processor.a")
    if(NOT EXISTS "${VAD_ARCHIVE}")
        message(FATAL_ERROR "Run firmware/setup_esp_sr.ps1 to fetch the pinned ESP-SR dependency.")
    endif()
    execute_process(COMMAND git -C "${ESP_SR_ROOT}" rev-parse HEAD
                    OUTPUT_VARIABLE ESP_SR_REV OUTPUT_STRIP_TRAILING_WHITESPACE
                    RESULT_VARIABLE ESP_SR_GIT_RESULT)
    if(NOT ESP_SR_GIT_RESULT EQUAL 0 OR NOT ESP_SR_REV STREQUAL "b6774a31118de04742e51ab57dd68c9001fa1091")
        message(FATAL_ERROR "ESP-SR revision differs from the reviewed PixelPet dependency.")
    endif()
    file(SHA256 "${VAD_ARCHIVE}" VAD_ARCHIVE_SHA)
    if(NOT VAD_ARCHIVE_SHA STREQUAL "ba2e2a2ff9bf14d87efe03b65a4b97877c823c7a3c2fcf7ecc115c6ada43a0e0")
        message(FATAL_ERROR "The ESP-SR VAD archive does not match the pinned dependency.")
    endif()
    idf_component_register(INCLUDE_DIRS "${ESP_SR_ROOT}/include/esp32s3")
    target_link_libraries(${COMPONENT_LIB} INTERFACE "${VAD_ARCHIVE}")
else()
    idf_component_register()
endif()
